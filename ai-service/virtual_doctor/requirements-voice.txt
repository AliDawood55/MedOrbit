# Virtual Doctor — Track B voice dependencies (speech-to-text + text-to-speech).
#
# Kept separate from the repo-root requirements.txt so this module stays
# additive and removable. Install with:
#
#     pip install -r ai-service/virtual_doctor/requirements-voice.txt
#
# No system packages are needed — in particular ffmpeg is NOT required.
# faster-whisper decodes audio through PyAV, whose wheels bundle their own
# FFmpeg libraries, so a plain pip install is enough on Windows, macOS and
# Linux alike.
#
# Pulls in ctranslate2 (inference engine), av (audio decoding),
# onnxruntime (Silero VAD), tokenizers and huggingface_hub automatically.
faster-whisper==1.2.1

# Already present in the repo-root requirements.txt, repeated here because the
# /virtual-doctor/transcribe endpoint receives multipart file uploads and will
# fail at import time without it.
python-multipart>=0.0.20

# Text-to-speech. Reuses the onnxruntime that faster-whisper already installs
# and bundles its own espeak-ng data, so it needs no system packages either —
# the historical piper-phonemize wheel problem on Windows does not apply to
# 1.6.0. Voices (~60MB each) download on first use into assets/tts_voices/.
#
# LICENSE: piper-tts is GPL-3.0 from 1.3.0 onwards because it links espeak-ng.
# Open source as required, but stronger copyleft than the rest of this repo —
# relevant only if MedOrbit is ever distributed as a binary.
piper-tts==1.6.0
